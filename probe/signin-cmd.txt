script:
@echo off
rem ClaudeOS Setup's sign-in window (opened without administrator rights).
title Sign in to Claude
if not "%~1"=="here" (
  where wt.exe >nul 2>nul && (
    start "" wt.exe -d "%~dp0." cmd /c signin.cmd here
    exit /b 0
  )
)
cd /d "%USERPROFILE%"
set "CLAUDE_CONFIG_DIR=%TEMP%\claudeos-signin-probe\config"
echo Your browser opens on Claude's sign-in page. Sign in with the account that has your Claude
echo subscription, and approve. ClaudeOS Setup carries on by itself.
echo If the browser shows a code instead, paste it here: Ctrl+V or a right-click.
echo.
"%TEMP%\claudeos-signin-probe\fake claude.cmd" auth login --claudeai
if exist "%TEMP%\claudeos-signin-probe\config\.credentials.json" (
  echo.
  echo Signed in. This window closes in a moment.
  timeout /t 4 >nul
) else (
  echo.
  echo The sign-in did not finish. Close this window: ClaudeOS Setup offers to try again.
  pause
)

ran:
Your browser opens on Claude's sign-in page. Sign in with the account that has your Claude
subscription, and approve. ClaudeOS Setup carries on by itself.
If the browser shows a code instead, paste it here: Ctrl+V or a right-click.

args: auth login --claudeai

saved a sign-in: True
